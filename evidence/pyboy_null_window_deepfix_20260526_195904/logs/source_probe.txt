    25	def classify_png_health(png: bytes) -> str:
    26	    """Return 'ok', 'too_small_or_blank', 'all_white', or 'all_black'.
    27	
    28	    Pure image-health check — does NOT judge game progress.
    29	    """
    30	    if not png or len(png) < 1000:
    31	        return "too_small_or_blank"
    32	    try:
    33	        img = Image.open(io.BytesIO(png)).convert("RGB")
    34	    except Exception:
    35	        return "too_small_or_blank"
    36	    pixels = list(img.getdata())
    37	    n = len(pixels)
    38	    if n == 0:
    39	        return "too_small_or_blank"
    40	    white = sum(1 for p in pixels if p == (255, 255, 255))
    41	    black = sum(1 for p in pixels if p == (0, 0, 0))
    42	    if white / n > 0.98:
    43	        return "all_white"
    44	    if black / n > 0.98:
    45	        return "all_black"
    46	    return "ok"

   227	    def tick(self, frames: int = 1) -> None:
   228	        """Advance emulation by *frames* frames.
   229	
   230	        Injects a vblank-interrupt pending flag (IF bit 0) before each
   231	        sub-frame tick and forces the PPU/LCD pipeline by reading the
   232	        screen buffer after each tick (only when the LCD is enabled).
   233	        PyBoy null-window mode does not generate real vblank or STAT
   234	        interrupts (no PPU scanlines), so Gen 1 dialog/cutscene text
   235	        cannot advance without these two steps.
   236	
   237	        The LCD-enable check (LCDC.7) prevents a renderer state corruption
   238	        during periods when the game briefly disables the LCD — for example
   239	        during battle attack animation transitions where the game switches
   240	        VRAM data. If ``screen.ndarray`` is read while the LCD is off, the
   241	        renderer produces a white frame and may fail to resume correctly
   242	        when the LCD is re-enabled (``battle_nullwindow_freeze``).
   243	        """
   244	        pb = self._pyboy
   245	        for _ in range(frames):
   246	            lcdc_before = pb.memory[0xFF40]
   247	            lcd_enabled = lcdc_before & 0x80
   248	
   249	            if lcd_enabled:
   250	                # LCD is running — inject fake vblank to advance the game's
   251	                # interrupt-driven subsystems (dialog/cutscene text, joypad
   252	                # polling, OAM DMA), then force the PPU pipeline via the
   253	                # screen buffer read so LY/STAT advance correctly in
   254	                # null-window mode.
   255	                pb.memory[0xFF0F] = pb.memory[0xFF0F] | 0x01
   256	                pb.tick()
   257	                _ = pb.screen.ndarray
   258	            else:
   259	                pb.tick()
   260	
   261	            # After the tick, check whether the game re-enabled the LCD
   262	            # during this frame.  Gen 1's battle engine briefly disables
   263	            # the LCD to load VRAM data (attack animation, HP bar, etc.),
   264	            # then turns it back on.  The re-enable event itself does not
   265	            # generate a vblank, so we inject one here and force a render
   266	            # to kickstart the scanline pipeline.
   267	            if not lcd_enabled:
   268	                lcdc_after = pb.memory[0xFF40]
   269	                if lcdc_after & 0x80:
   270	                    pb.memory[0xFF0F] = pb.memory[0xFF0F] | 0x01
   271	                    _ = pb.screen.ndarray
   272	
   273	            self.frame_count += 1
